cmake_minimum_required(VERSION 3.22)
project(makeuphouse_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT LLAMA_SOURCE_DIR)
 include(FetchContent)
 FetchContent_Declare(llama_source URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/b1b770d90d94cad9e9ea5b993bdbc383a522fb28 URL_HASH SHA256=ae8556670b17c25931cd39eb7c95617650eaf6091bbcff206750583ddde1acb4)
 FetchContent_GetProperties(llama_source)
 if(NOT llama_source_POPULATED)
  FetchContent_Populate(llama_source)
 endif()
 set(LLAMA_SOURCE_DIR ${llama_source_SOURCE_DIR})
endif()
add_subdirectory(${LLAMA_SOURCE_DIR} llama-build)
add_library(mh_runtime STATIC runtime.cpp vision.cpp)
target_include_directories(mh_runtime PUBLIC .)
target_include_directories(mh_runtime PUBLIC ${LLAMA_SOURCE_DIR}/tools/mtmd)
target_link_libraries(mh_runtime PUBLIC llama mtmd)
if(ANDROID)
 add_library(mh_local_ai SHARED ../android/app/src/main/cpp/bridge.cpp)
 target_link_libraries(mh_local_ai PRIVATE mh_runtime log android)
 target_link_options(mh_local_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(mh-ai-poc poc.cpp)
 target_link_libraries(mh-ai-poc PRIVATE mh_runtime)
 add_executable(mh-ai-session session.cpp)
 target_include_directories(mh-ai-session PRIVATE ${LLAMA_SOURCE_DIR}/vendor)
 target_link_libraries(mh-ai-session PRIVATE mh_runtime)
 add_executable(mh-ai-constraints-test constraints_test.cpp)
 add_executable(mh-ai-vision-bounds ../scripts/vision-probe-boundaries.cpp)
 target_link_libraries(mh-ai-vision-bounds PRIVATE mh_runtime)
 add_executable(mh-ai-test runtime_test.cpp)
 target_link_libraries(mh-ai-test PRIVATE mh_runtime)
endif()
